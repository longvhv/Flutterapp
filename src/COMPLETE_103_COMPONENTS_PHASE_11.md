## 💻 Usage Examples

### VHVRow - Horizontal Layout
```tsx
<VHVRow 
  mainAxisAlignment="spaceBetween"
  crossAxisAlignment="center"
>
  <Text>Left</Text>
  <Icon />
  <Button>Right</Button>
</VHVRow>

<VHVRow mainAxisAlignment="spaceEvenly">
  <Item1 /> <Item2 /> <Item3 />
</VHVRow>
```

### VHVColumn - Vertical Layout
```tsx
<VHVColumn 
  mainAxisAlignment="center"
  crossAxisAlignment="stretch"
>
  <Header />
  <Content />
  <Footer />
</VHVColumn>

<VHVColumn mainAxisSize="min">
  <CompactContent />
</VHVColumn>
```

### VHVCenter - Simple Centering
```tsx
<VHVCenter>
  <Logo />
</VHVCenter>

<VHVCenter widthFactor={0.8} heightFactor={0.5}>
  <Dialog />
</VHVCenter>
```

### VHVAlign - Custom Alignment
```tsx
<VHVAlign alignment="topRight">
  <CloseButton />
</VHVAlign>

<VHVAlign alignment="bottomCenter">
  <BottomAction />
</VHVAlign>
```

### VHVPadding - Simple Padding
```tsx
<VHVPadding padding={16}>
  <Content />
</VHVPadding>

<VHVPadding padding={{ horizontal: 20, vertical: 10 }}>
  <Card />
</VHVPadding>

<VHVPadding padding={{ top: 10, left: 20, right: 20, bottom: 30 }}>
  <CustomContent />
</VHVPadding>
```

### VHVContainer - The Swiss Army Knife
```tsx
<VHVContainer
  width={300}
  height={200}
  padding={20}
  margin={10}
  color="#3B82F6"
  borderRadius={16}
  border={{ color: '#1E40AF', width: 2 }}
  boxShadow="0 4px 6px rgba(0,0,0,0.1)"
  alignment="center"
>
  <Content />
</VHVContainer>

<VHVContainer
  gradient={{
    colors: ['#3B82F6', '#8B5CF6'],
    begin: 'to right',
  }}
  borderRadius={{ topLeft: 16, topRight: 16 }}
>
  <GradientCard />
</VHVContainer>
```

### VHVDecoratedBox - Box Decoration
```tsx
<VHVDecoratedBox
  decoration={{
    gradient: {
      colors: ['#FF6B6B', '#FFE66D'],
      begin: 'to bottom right',
    },
    borderRadius: 12,
    boxShadow: '0 8px 16px rgba(0,0,0,0.2)',
  }}
>
  <BeautifulCard />
</VHVDecoratedBox>

<VHVDecoratedBox
  decoration={{
    image: {
      url: 'background.jpg',
      fit: 'cover',
      opacity: 0.3,
    },
    color: '#000000',

  position="background"
>
  <OverlayContent />
</VHVDecoratedBox>
```
