### VHVFutureBuilder - Async Data
```tsx
<VHVFutureBuilder
  future={fetchUserData()}
  builder={(snapshot) => {
    if (snapshot.connectionState === 'waiting') {
      return <VHVLoader />;
    }
    if (snapshot.hasError) {
      return <VHVText>Error: {snapshot.error}</VHVText>;
    }
    if (snapshot.hasData) {
      return <UserProfile data={snapshot.data} />;
    }
    return <VHVEmptyState />;
  }}
/>
```

### VHVStreamBuilder - Real-time Streams
```tsx
<VHVStreamBuilder
  stream={messageStream}
  builder={(snapshot) => {
    if (snapshot.hasData) {
      return <MessageList messages={snapshot.data} />;
    }
    return <VHVLoader />;
  }}
/>
```
